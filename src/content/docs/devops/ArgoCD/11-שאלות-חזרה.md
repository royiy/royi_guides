שאלות חזרה אינטראקטיביות על כל מה שלמדתם לאורך מדריכי ה-ArgoCD (מבוא ל-GitOps, התקנה, ארכיטקטורה ורכיבים, יצירת Application ראשונה, Projects/RBAC, Sync Policies/Hooks/Health Checks, ApplicationSets ו-Multi-Cluster, אינטגרציה עם Helm/Kustomize, Best Practices, ושאלות ראיון עבודה). בחרו תשובה בכל שאלה ותקבלו מיד משוב האם היא נכונה, יחד עם הסבר.

<div class="quiz-card" data-answer="ב">
  <p class="quiz-question"><strong>1. מהו ההבדל המרכזי בין גישת GitOps מבוססת Pull (כפי ש-ArgoCD מיישם) לבין CI/CD מסורתי מבוסס Push?</strong></p>
  <div class="quiz-options">
    <button class="quiz-option" type="button" data-choice="א">א. אין הבדל מעשי בין השתיים - שתי הגישות זהות בפועל</button>
    <button class="quiz-option" type="button" data-choice="ב">ב. ב-Push-based כלי ה-CI/CD (כמו Jenkins) דוחף שינויים ישירות ל-Cluster ולכן צריך הרשאות אליו, ואילו ב-Pull-based (GitOps) סוכן כמו ArgoCD יושב בתוך ה-Cluster ומושך שינויים מ-Git, כך שאין צורך לתת לכלי חיצוני הרשאות Cluster</button>
    <button class="quiz-option" type="button" data-choice="ג">ג. Push-based עובד רק עם Helm, ו-Pull-based עובד רק עם YAML גולמי</button>
    <button class="quiz-option" type="button" data-choice="ד">ד. ב-GitOps אסור להשתמש ב-<code>kubectl</code> בכלל, בשום שלב</button>
  </div>
  <div class="quiz-result" aria-live="polite"></div>
  <div class="quiz-explain"><strong>הסבר:</strong> בגישה המסורתית ה-CI/CD צריך הרשאות ישירות ל-Cluster כדי לבצע <code>kubectl apply</code> - סיכון אבטחתי, וגם אין מנגנון אוטומטי לזיהוי Drift. ב-GitOps, ArgoCD "מושך" את השינויים מבפנים, כך שה-Cluster אחראי על עצמו ואין צורך לחשוף Credentials לכלי חיצוני.</div>
</div>

<div class="quiz-card" data-answer="ב">
  <p class="quiz-question"><strong>2. בפקודת ההתקנה <code>kubectl apply -n argocd --server-side --force-conflicts -f .../install.yaml</code>, למה נדרשים הדגלים <code>--server-side --force-conflicts</code>?</strong></p>
  <div class="quiz-options">
    <button class="quiz-option" type="button" data-choice="א">א. כדי להתקין דווקא את גרסת ה-HA של ArgoCD</button>
    <button class="quiz-option" type="button" data-choice="ב">ב. כי חלק מה-CRDs של ArgoCD חורגים מהגודל המקסימלי שנתמך ב-Apply הרגיל (Client-Side)</button>
    <button class="quiz-option" type="button" data-choice="ג">ג. כדי לאפשר חיבור SSO מול GitHub באופן אוטומטי</button>
    <button class="quiz-option" type="button" data-choice="ד">ד. כדי למחוק את ה-Namespace הישן לפני ההתקנה</button>

  <div class="quiz-explain"><strong>הסבר:</strong> כפי שצוין במדריך ההתקנה, חלק מה-CRDs של ArgoCD גדולים מדי עבור Apply רגיל (Client-Side), ולכן חובה להשתמש ב-<code>--server-side</code> יחד עם <code>--force-conflicts</code> כדי שההתקנה תצליח.</div>

<div class="quiz-card" data-answer="ב">
  <p class="quiz-question"><strong>3. מהו התפקיד של הרכיב <code>argocd-application-controller</code> בארכיטקטורת ArgoCD?</strong></p>
  <div class="quiz-options">

    <button class="quiz-option" type="button" data-choice="ג">ג. הוא אחראי אך ורק על אימות משתמשים מול SSO/OIDC</button>

  </div>
  <div class="quiz-result" aria-live="polite"></div>
  <div class="quiz-explain"><strong>הסבר:</strong> ה-<code>application-controller</code> הוא הרכיב שמריץ את לולאת ה-Reconciliation, משווה בין המניפסטים שמגיעות מה-repo-server (Desired) לבין המצב בפועל ב-Cluster (Live), מעדכן את שדות ה-<code>status.sync</code> וה-<code>status.health</code>, ומבצע את ה-Sync בפועל. אימות SSO הוא תפקידו של <code>argocd-dex-server</code>, ושכפול ה-Repo הוא תפקידו של <code>argocd-repo-server</code>.</div>
</div>

<div class="quiz-card" data-answer="ב">

  <div class="quiz-options">

    <button class="quiz-option" type="button" data-choice="ג">ג. תבנית RBAC שנותנת הרשאות Admin אוטומטיות לכל המשתמשים</button>

</div>

<div class="quiz-card" data-answer="ב">
  <p class="quiz-question"><strong>10. למה <code>argocd app rollback</code> לא מספיק כפתרון קבוע כשיש <code>automated sync</code> פעיל, ולמה עדיף במקום זאת <code>git revert</code>?</strong></p>
  <div class="quiz-options">
    <button class="quiz-option" type="button" data-choice="א">א. כי הפקודה <code>argocd app rollback</code> לא נתמכת יותר בגרסאות עדכניות של ArgoCD</button>
    <button class="quiz-option" type="button" data-choice="ב">ב. כי הפקודה משנה רק את המצב ב-Cluster באופן זמני, ואם <code>automated sync</code> פעיל ArgoCD יחזיר את המצב לגרסה העדכנית ב-Git בפעם הבאה שירוץ Reconciliation - <code>git revert</code> הוא ה"תיקון האמיתי" כי הוא משנה את מקור האמת (Git) עצמו</button>
    <button class="quiz-option" type="button" data-choice="ג">ג. כי <code>argocd app rollback</code> דורש תמיד הרשאות Cluster-Admin מלאות</button>
    <button class="quiz-option" type="button" data-choice="ד">ד. כי <code>rollback</code> עובד רק על Applications שנוצרו דרך Helm</button>
  </div>
  <div class="quiz-result" aria-live="polite"></div>
  <div class="quiz-explain"><strong>הסבר:</strong> Rollback דרך ArgoCD משנה רק את המצב ב-Cluster זמנית - Git עצמו לא משתנה. אם Automated Sync פעיל, ArgoCD "יתקן" את זה בחזרה לגרסה העדכנית ב-Git ברגע שירוץ Reconciliation הבא, מה שיבטל את ה-Rollback. הדרך הנכונה ב-GitOps אמיתי היא לבצע <code>git revert</code> על ה-Commit הבעייתי, כך שמקור האמת עצמו משקף את המצב הרצוי.</div>
</div>

(function () {
  function initQuiz() {
    var cards = document.querySelectorAll('.quiz-card');
    var scoreEl = document.getElementById('quiz-score');
    var resetAllBtn = document.getElementById('quiz-reset-all');

      var correctCount = document.querySelectorAll('.quiz-card--answered[data-was-correct="true"]').length;
      scoreEl.textContent = 'ענית נכון על ' + correctCount + ' מתוך ' + cards.length;

      var chosen = chosenBtn.getAttribute('data-choice');
      var isCorrect = chosen === correct;

      card.setAttribute('data-was-correct', isCorrect ? 'true' : 'false');

      var options = card.querySelectorAll('.quiz-option');
      options.forEach(function (opt) {
        opt.disabled = true;
        if (opt.getAttribute('data-choice') === correct) {
          opt.classList.add('quiz-option--correct');
        } else if (opt === chosenBtn) {
          opt.classList.add('quiz-option--incorrect');
        }
      });

      var result = card.querySelector('.quiz-result');
      if (result) {
        result.textContent = isCorrect ? '✅ נכון!' : '❌ לא נכון - התשובה הנכונה מסומנת בירוק.';
        result.classList.remove('quiz-result--correct', 'quiz-result--incorrect');
        result.classList.add(isCorrect ? 'quiz-result--correct' : 'quiz-result--incorrect');
      }

      updateScore();
    }

    function resetCard(card) {
      card.classList.remove('quiz-card--answered');
      card.removeAttribute('data-was-correct');
      var options = card.querySelectorAll('.quiz-option');
      options.forEach(function (opt) {
        opt.disabled = false;
        opt.classList.remove('quiz-option--correct', 'quiz-option--incorrect');
      });
      var result = card.querySelector('.quiz-result');
      if (result) {
        result.textContent = '';
        result.classList.remove('quiz-result--correct', 'quiz-result--incorrect');
      }
    }

    cards.forEach(function (card) {
      card.querySelectorAll('.quiz-option').forEach(function (btn) {
        btn.addEventListener('click', function () {
          if (card.classList.contains('quiz-card--answered')) return;
          answerCard(card, btn);
        });
      });
    });

    if (resetAllBtn) {
      resetAllBtn.addEventListener('click', function () {
        cards.forEach(resetCard);
        updateScore();
      });
    }

    updateScore();
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initQuiz);
  } else {
    initQuiz();
  }
})();
</script>
