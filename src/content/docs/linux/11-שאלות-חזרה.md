שאלות חזרה אינטראקטיביות על כל מה שלמדתם לאורך מדריכי הלינוקס (פקודות בסיסיות, ניהול קבצים, הרשאות, משתמשים וקבוצות, תהליכים, systemd, Networking, ניהול שרתים, Bash Scripting ושאלות ראיון). בחרו תשובה בכל שאלה ותקבלו מיד משוב האם היא נכונה, יחד עם הסבר.

<div class="quiz-card" data-answer="ב">
  <p class="quiz-question"><strong>1. מה ההבדל בין <code>&gt;</code> ל-<code>&gt;&gt;</code> בהפניית פלט (redirection)?</strong></p>
  <div class="quiz-options">
    <button class="quiz-option" type="button" data-choice="א">א. אין הבדל, שני הסימנים שקולים לחלוטין</button>
    <button class="quiz-option" type="button" data-choice="ב">ב. <code>&gt;</code> דורס את תוכן הקובץ, ו-<code>&gt;&gt;</code> מוסיף לסופו בלי למחוק את הקיים</button>
    <button class="quiz-option" type="button" data-choice="ג">ג. <code>&gt;</code> מפנה קלט לפקודה, ו-<code>&gt;&gt;</code> מפנה פלט</button>
    <button class="quiz-option" type="button" data-choice="ד">ד. <code>&gt;&gt;</code> עובד רק עם <code>grep</code></button>
  </div>
  <div class="quiz-result" aria-live="polite"></div>
  <div class="quiz-explain"><strong>הסבר:</strong> כפי שנלמד במדריך הפקודות הבסיסיות, <code>echo "Hello" &gt; output.txt</code> דורס את הקובץ (או יוצר אותו מחדש), בעוד <code>echo "World" &gt;&gt; output.txt</code> מוסיף שורה חדשה בסוף הקובץ הקיים בלי לפגוע בתוכן שכבר נמצא בו.</div>
</div>

<div class="quiz-card" data-answer="א">
  <p class="quiz-question"><strong>2. מהו ההבדל המרכזי בין Hard Link ל-Symbolic Link בלינוקס?</strong></p>
  <div class="quiz-options">
    <button class="quiz-option" type="button" data-choice="א">א. Hard Link הוא שם נוסף לאותו inode בדיסק (הקובץ נמחק בפועל רק כשכל ה-Hard Links נמחקים), בעוד Symbolic Link הוא קובץ נפרד שמצביע לנתיב, ונשבר אם המקור נמחק</button>
    <button class="quiz-option" type="button" data-choice="ב">ב. Symbolic Link יכול להצביע רק על קבצים בתוך אותה תיקייה</button>
    <button class="quiz-option" type="button" data-choice="ג">ג. Hard Link יכול להצביע על תיקיות, ואילו Symbolic Link לא</button>
    <button class="quiz-option" type="button" data-choice="ד">ד. אין הבדל מעשי בין השניים, זו רק מוסכמת שמות</button>
  </div>
  <div class="quiz-result" aria-live="polite"></div>
  <div class="quiz-explain"><strong>הסבר:</strong> כפי שהוסבר במדריך ניהול הקבצים המתקדם, Hard Link (<code>ln</code>) מצביע ישירות ל-inode ולכן אינו יכול להצביע על תיקייה, בעוד Symbolic Link (<code>ln -s</code>) הוא קובץ נפרד המכיל נתיב, יכול להצביע גם על תיקיות ולעבור בין מערכות קבצים - אך נשבר אם הקובץ המקורי נמחק.</div>
</div>

<div class="quiz-card" data-answer="ב">
  <p class="quiz-question"><strong>3. מה עושה סיבית ה-SUID (למשל <code>chmod 4755 /usr/bin/passwd</code>) על קובץ הרצה?</strong></p>
  <div class="quiz-options">

    <button class="quiz-option" type="button" data-choice="ג">ג. יוצרת קישור סימבולי אוטומטי לקובץ</button>
    <button class="quiz-option" type="button" data-choice="ד">ד. מצפינה את תוכן הקובץ בדיסק</button>
  </div>
  <div class="quiz-result" aria-live="polite"></div>
  <div class="quiz-explain"><strong>הסבר:</strong> במדריך ההרשאות והבעלות הוסבר ש-SUID (ספרה 4 בשיטה האוקטלית, מוצג כ-<code>rws</code> בפלט <code>ls -l</code>) גורם לתוכנית לרוץ עם הרשאות הבעלים שלה. זו הסיבה ש-<code>passwd</code> יכול לכתוב לקובץ <code>/etc/shadow</code> שנגיש רק ל-root, גם כשמריץ אותו משתמש רגיל - וגם למה SUID על קובץ פגיע הוא סיכון אבטחה חמור.</div>
</div>

<div class="quiz-card" data-answer="ג">

    <button class="quiz-option" type="button" data-choice="ג">ג. <code>-aG</code> (append) מוסיף את המשתמש לקבוצת developers בלי לפגוע בקבוצות המשניות הקיימות שלו, בעוד <code>-G</code> לבדו מחליף את כל הקבוצות המשניות ומשאיר רק את developers</button>
    <button class="quiz-option" type="button" data-choice="ד">ד. <code>-G</code> משנה את הקבוצה הראשית, ו-<code>-aG</code> משנה רק סיסמה</button>

SIGKILL (9) הוא מוצא אחרון - הרג מיידי שלא ניתן לתפוס או להתעלם ממנו, ולכן אינו מאפשר לתהליך לסיים בצורה מסודרת.</div>

    <button class="quiz-option" type="button" data-choice="ג">ג. <code>restart</code> זמין רק ל-root, ו-<code>reload</code> לכל משתמש</button>
    <button class="quiz-option" type="button" data-choice="ד">ד. אין הבדל בפועל בין הפקודות</button>
  </div>
  <div class="quiz-result" aria-live="polite"></div>
  <div class="quiz-explain"><strong>הסבר:</strong> מדריך ה-systemd מבהיר ש-<code>restart</code> גורם להפרעה זמנית כי הוא עוצר ומפעיל את התהליך מחדש לגמרי, בעוד <code>reload</code> מאפשר לשירותים התומכים בכך (כמו nginx) לטעון קונפיגורציה חדשה תוך כדי ריצה, בלי לנתק חיבורים פעילים.</div>
</div>

<script>
(function () {
  function initQuiz() {
    var cards = document.querySelectorAll('.quiz-card');
    var scoreEl = document.getElementById('quiz-score');
    var resetAllBtn = document.getElementById('quiz-reset-all');

    function updateScore() {
      if (!scoreEl) return;
      var correctCount = document.querySelectorAll('.quiz-card--answered[data-was-correct="true"]').length;
      scoreEl.textContent = 'ענית נכון על ' + correctCount + ' מתוך ' + cards.length;
    }

    function answerCard(card, chosenBtn) {
      var correct = card.getAttribute('data-answer');
      var chosen = chosenBtn.getAttribute('data-choice');
      var isCorrect = chosen === correct;

      card.classList.add('quiz-card--answered');
      card.setAttribute('data-was-correct', isCorrect ? 'true' : 'false');

      var options = card.querySelectorAll('.quiz-option');
      options.forEach(function (opt) {
        opt.disabled = true;
        if (opt.getAttribute('data-choice') === correct) {
          opt.classList.add('quiz-option--correct');
        } else if (opt === chosenBtn) {
          opt.classList.add('quiz-option--incorrect');
        }
      });

      var result = card.querySelector('.quiz-result');
      if (result) {
        result.textContent = isCorrect ? '✅ נכון!' : '❌ לא נכון - התשובה הנכונה מסומנת בירוק.';
        result.classList.remove('quiz-result--correct', 'quiz-result--incorrect');
        result.classList.add(isCorrect ? 'quiz-result--correct' : 'quiz-result--incorrect');
      }

      updateScore();
    }

    function resetCard(card) {
      card.classList.remove('quiz-card--answered');
      card.removeAttribute('data-was-correct');
      var options = card.querySelectorAll('.quiz-option');
      options.forEach(function (opt) {
        opt.disabled = false;
        opt.classList.remove('quiz-option--correct', 'quiz-option--incorrect');
      });
      var result = card.querySelector('.quiz-result');
      if (result) {
        result.textContent = '';
        result.classList.remove('quiz-result--correct', 'quiz-result--incorrect');
      }
    }

    cards.forEach(function (card) {
      card.querySelectorAll('.quiz-option').forEach(function (btn) {
        btn.addEventListener('click', function () {
          if (card.classList.contains('quiz-card--answered')) return;
          answerCard(card, btn);
        });
      });
    });

    if (resetAllBtn) {
      resetAllBtn.addEventListener('click', function () {
        cards.forEach(resetCard);
        updateScore();
      });
    }

    updateScore();
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initQuiz);
  } else {
    initQuiz();
  }
})();
</script>
